2) Fetch all the dept details with more than 1 employee
SELECT d.*FROM Dept d
JOIN Emp e ON d.DeptNo = e.DeptNo
GROUP BY d.DeptNo, d.Dname, d.Loc
HAVING COUNT(e.EmpNo) > 1;

25) List manager having maximum number of employees
SELECT TOP 1 m.Ename AS Manager_Name,
       COUNT(e.EmpNo) AS Employee_Count
FROM Emp e JOIN Emp m
ON e.Mgr = m.EmpNo
GROUP BY m.Ename
ORDER BY COUNT(e.EmpNo) DESC;


26) Employee, department and manager details of employee 
    who has second highest salary
SELECT e.*, d.*, m.Ename AS Manager_Name
FROM Emp e
LEFT JOIN Dept d
ON e.DeptNo = d.DeptNo
LEFT JOIN Emp m
ON e.Mgr = m.EmpNo
WHERE e.Sal = (
    SELECT MAX(Sal)
    FROM Emp
    WHERE Sal < (SELECT MAX(Sal) FROM Emp)
);


27) List all details of all managers
SELECT *
FROM Emp
WHERE EmpNo IN (
    SELECT DISTINCT Mgr
    FROM Emp
    WHERE Mgr IS NOT NULL
);

29) Employees who are managers, take commission less than 1000
    and work in Delhi
SELECT e.*
FROM Emp e
JOIN Dept d
ON e.DeptNo = d.DeptNo
WHERE e.EmpNo IN (
    SELECT DISTINCT Mgr
    FROM Emp
    WHERE Mgr IS NOT NULL
)
AND e.Commission < 1000
AND d.Loc = 'Delhi';
